-- Query : Q4 — Unresolved Expiry Alerts with Blood Unit Details
-- Purpose: Monitoring dashboard — shows all active (unresolved) expiry alerts
--          for a blood bank, joined with the blood unit's remaining shelf life.
--          Used by blood bank staff to prioritise redistribution before wastage.
-- Index Used: expiry_alerts_active_unit_unique_idx (partial unique)
--             expiry_alerts_institution_unresolved_idx
--             blood_units_expiry_fefo_idx
-- =============================================================================

EXPLAIN (ANALYZE, BUFFERS, FORMAT TEXT)
SELECT
    ea.id               AS alert_id,
    ea.severity,
    ea.created_at       AS alerted_at,
    bu.unit_code,
    bu.component_type,
    bu.abo_group,
    bu.rh_factor,
    bu.expiry_date,
    EXTRACT(EPOCH FROM (bu.expiry_date - NOW())) / 3600 AS hours_remaining
FROM expiry_alerts ea
JOIN blood_units bu ON bu.id = ea.blood_unit_id
WHERE
    ea.institution_id = 'd4e5f6a7-b8c9-0123-def0-234567890123'
    AND ea.resolved   = false
ORDER BY bu.expiry_date ASC;
