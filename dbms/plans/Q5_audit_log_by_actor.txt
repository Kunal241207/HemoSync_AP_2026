-- Query : Q5 — Audit Log — Recent Actions by a Specific User
-- Purpose: Compliance and security audit trail lookup. Retrieves the last 50
--          actions performed by a specific actor within a given institution,
--          ordered by recency.
-- Index Used: audit_logs_actor_time_idx (actor_id + created_at DESC)
-- =============================================================================

EXPLAIN (ANALYZE, BUFFERS, FORMAT TEXT)
SELECT
    al.id,
    al.entity_type,
    al.entity_id,
    al.action,
    al.metadata,
    al.created_at
FROM audit_logs al
WHERE
    al.actor_id       = 'e5f6a7b8-c9d0-1234-ef01-345678901234'
    AND al.institution_id = 'd4e5f6a7-b8c9-0123-def0-234567890123'
ORDER BY al.created_at DESC
LIMIT 50;
