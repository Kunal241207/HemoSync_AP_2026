-- Purpose: Core blood matching lookup used by the allocation service.
--          Fetches all AVAILABLE, tested, released units of a specific
--          component type, ABO group, and Rh factor at an institution,
--          ordered by expiry date (FEFO — First Expired, First Out).
-- Index Used: blood_units_eligibility_lookup_idx (composite partial)

EXPLAIN (ANALYZE, BUFFERS, FORMAT TEXT)
SELECT
    bu.id,
    bu.unit_code,
    bu.component_type,
    bu.abo_group,
    bu.rh_factor,
    bu.volume_ml,
    bu.expiry_date
FROM blood_units bu
WHERE
    bu.institution_id    = 'a1b2c3d4-e5f6-7890-abcd-ef1234567890'
    AND bu.component_type  = 'RED_BLOOD_CELLS'
    AND bu.abo_group       = 'O'
    AND bu.rh_factor       = 'POSITIVE'
    AND bu.inventory_status = 'AVAILABLE'
    AND bu.testing_status   = 'PASSED'
    AND bu.release_status   = 'RELEASED'
    AND bu.expiry_date      > NOW()
ORDER BY bu.expiry_date ASC;
