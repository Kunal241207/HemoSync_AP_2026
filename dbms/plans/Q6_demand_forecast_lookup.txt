-- Query : Q6 — Demand Forecast Retrieval for Upcoming 7-Day Horizon
-- Purpose: Intelligence dashboard — retrieves the latest demand forecast series
--          for each blood component type at an institution for the next 7 days,
--          ranked by most recently generated model run.
-- Index Used: demand_forecasts_lookup_idx
--             demand_forecasts_series_unique_idx (implicit ordering hint)
-- =============================================================================

EXPLAIN (ANALYZE, BUFFERS, FORMAT TEXT)
SELECT DISTINCT ON (df.component_type, df.abo_group, df.rh_factor)
    df.component_type,
    df.abo_group,
    df.rh_factor,
    df.forecast_date,
    df.horizon_days,
    df.predicted_demand,
    df.lower_bound,
    df.upper_bound,
    df.model_version,
    df.metrics,
    df.generated_at
FROM demand_forecasts df
WHERE
    df.institution_id  = 'd4e5f6a7-b8c9-0123-def0-234567890123'
    AND df.horizon_days = 7
    AND df.forecast_date >= CURRENT_DATE
    AND df.forecast_date <  CURRENT_DATE + INTERVAL '7 days'
ORDER BY
    df.component_type, df.abo_group, df.rh_factor,
    df.generated_at DESC;
