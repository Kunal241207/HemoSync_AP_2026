-- Query : Q2 — List Pending / Urgent Blood Requests for a Hospital
-- Purpose: Dashboard query showing open blood requests ordered by urgency
--          and required_by deadline. Used by hospital staff to prioritize
--          procurement workflows.
-- Index Used: blood_requests_hospital_status_idx
--             blood_requests_required_by_urgency_idx

EXPLAIN (ANALYZE, BUFFERS, FORMAT TEXT)
SELECT
    br.id,
    br.requested_component,
    br.requested_abo,
    br.requested_rh,
    br.requested_quantity,
    br.urgency,
    br.status,
    br.required_by,
    br.created_at,
    u.name  AS requested_by
FROM blood_requests br
JOIN users u ON u.id = br.created_by_id
WHERE
    br.hospital_id = 'b2c3d4e5-f6a7-8901-bcde-f12345678901'
    AND br.status IN ('PENDING', 'VALIDATED', 'PARTIALLY_ALLOCATED')
ORDER BY
    CASE br.urgency
        WHEN 'EMERGENCY' THEN 1
        WHEN 'URGENT'    THEN 2
        ELSE                  3
    END,
    br.required_by ASC;
