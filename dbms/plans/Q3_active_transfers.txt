-- Query : Q3 — Active Transfers Destined for a Facility (with Unit Count)
-- Purpose: Logistics dashboard — shows all in-flight blood shipments en route
--          to a facility with aggregated unit counts per transfer.
--          Used by receiving facility staff to plan cold-chain reception.
-- Index Used: transfers_dest_status_idx
--             transfer_units_blood_unit_id_idx
-- =============================================================================

EXPLAIN (ANALYZE, BUFFERS, FORMAT TEXT)
SELECT
    t.id               AS transfer_id,
    t.status,
    t.dispatched_at,
    t.received_at,
    i_src.name         AS source_institution,
    COUNT(tu.id)       AS unit_count,
    t.notes
FROM transfers t
JOIN institutions i_src ON i_src.id = t.source_institution_id
LEFT JOIN transfer_units tu ON tu.transfer_id = t.id
WHERE
    t.destination_institution_id = 'c3d4e5f6-a7b8-9012-cdef-123456789012'
    AND t.status IN ('APPROVED', 'DISPATCHED')
GROUP BY
    t.id, t.status, t.dispatched_at, t.received_at, i_src.name, t.notes
ORDER BY t.dispatched_at DESC NULLS LAST;
